def ejecutar_pipeline_saneamiento(df_in: pd.DataFrame) -> pd.DataFrame:
    print("="*80)
    print("🚀 INICIANDO EJECUCIÓN DEL PIPELINE DE INGENIERÍA DE DATOS (CAPA SILVER)")
    print("="*80)
    
    df = df_in.copy()
    filas_raw = len(df)
    
    # --------------------------------------------------------------------------
    # 1. TRATAMIENTO DE DUPLICADOS
    # --------------------------------------------------------------------------
    df = df.drop_duplicates()
    duplicados_filas = filas_raw - len(df)
    
    # Resolver duplicados en la clave primaria id_transaccion conservando la primera aparición
    duplicados_id = df.duplicated(subset=['id_transaccion']).sum()
    df = df.drop_duplicates(subset=['id_transaccion'], keep='first')
    print(f"🧹 [DEDUPLICACIÓN] Filas exactas eliminadas: {duplicados_filas} | Claves 'id_transaccion' duplicadas eliminadas: {duplicados_id}")

    # --------------------------------------------------------------------------
    # 2. LIMPIEZA DE ERRORES TIPOGRÁFICOS Y ESPACIOS EN 'concepto' Y 'proveedor'
    # --------------------------------------------------------------------------
    # Reemplazo de caracteres Leet Speak comunes
    leet_dict = {'4': 'a', '3': 'e', '1': 'i', '0': 'o', '5': 's'}
    
    def limpiar_texto(val):
        if pd.isna(val):
            return np.nan
        text = str(val).strip() # Trimming de espacios sobrantes
        for k, v in leet_dict.items():
            text = text.replace(k, v)
        # Normalizar espacios múltiples internos y convertir a Formato Título
        text = re.sub(r'\s+', ' ', text)
        return text.title()

    df['concepto_clean'] = df['concepto'].apply(limpiar_texto)
    df['proveedor_clean'] = df['proveedor'].apply(limpiar_texto)
    
    # --------------------------------------------------------------------------
    # 3. CONVERSIÓN DE 'monto_usd' Y RECTIFICACIÓN DE SIGNOS/FORMATOS
    # --------------------------------------------------------------------------
    def limpiar_monto(val):
        if pd.isna(val):
            return np.nan
        # Eliminar cualquier caracter que no sea dígito, punto o signo negativo
        val_clean = re.sub(r'[^\d.-]', '', str(val))
        try:
            monto = float(val_clean)
            return abs(monto) # Corregir montos con signos negativos erróneos
        except ValueError:
            return np.nan

    df['monto_usd_clean'] = df['monto_usd'].apply(limpiar_monto)

    # --------------------------------------------------------------------------
    # 4. PARSEO ROBUSTO DE FECHAS ('fecha_gasto')
    # --------------------------------------------------------------------------
    fechas_normalizadas = df['fecha_gasto'].astype(str).str.replace('.', '/', regex=False)
    df['fecha_gasto_clean'] = pd.to_datetime(fechas_normalizadas, errors='coerce', format='mixed')
    
    # Imputación de fechas faltantes usando propagación temporal (ffill/bfill)
    df['fecha_gasto_clean'] = df['fecha_gasto_clean'].ffill().bfill()

    # --------------------------------------------------------------------------
    # 5. TRATAMIENTO DE VALORES NULOS (NaN) POR MEDIANA AGRUPADA
    # --------------------------------------------------------------------------
    # Imputar monto_usd faltante usando la mediana del concepto; si es nulo, la mediana global
    medianas_concepto = df.groupby('concepto_clean')['monto_usd_clean'].transform('median')
    mediana_global = df['monto_usd_clean'].median()
    
    df['monto_usd_clean'] = df['monto_usd_clean'].fillna(medianas_concepto).fillna(mediana_global)
    df['concepto_clean'] = df['concepto_clean'].fillna("Gasto No Especificado")
    df['metodo_pago_clean'] = df['metodo_pago'].fillna("No Registrado").astype(str).str.strip().str.title()
    df['estado_pago_clean'] = df['estado_pago'].fillna("PENDIENTE").astype(str).str.upper().str.strip()
    df['estado_pago_clean'] = df['estado_pago_clean'].replace({'PAGADO': 'APROBADO', 'PAGADO': 'APROBADO'})

    # --------------------------------------------------------------------------
    # 6. TRATAMIENTO ESTADÍSTICO DE OUTLIERS (IQR & Z-SCORE ROBUSTO)
    # --------------------------------------------------------------------------
    # A. Z-Score Robusto basado en MAD
    mediana_monto = df['monto_usd_clean'].median()
    mad_monto = np.median(np.abs(df['monto_usd_clean'] - mediana_monto))
    # Para distribución normal equivalente: 0.6745 * (x - mediana) / MAD
    df['z_score_robusto'] = 0.6745 * (df['monto_usd_clean'] - mediana_monto) / (mad_monto + 1e-6)
    
    # B. Rango Intercuartílico (IQR)
    Q1 = df['monto_usd_clean'].quantile(0.25)
    Q3 = df['monto_usd_clean'].quantile(0.75)
    IQR = Q3 - Q1
    limite_superior = Q3 + 1.5 * IQR
    
    # Detección de Outliers
    outliers_mask = (df['z_score_robusto'].abs() > 3.5) | (df['monto_usd_clean'] > limite_superior)
    print(f"⚠️ [ANOMALÍAS] Outliers detectados por Z-Score Robusto / IQR: {outliers_mask.sum()} registros.")
    
    # Cap / Acotamiento de outliers extremos a la frontera del límite superior
    df['monto_usd_final'] = np.where(df['monto_usd_clean'] > limite_superior, limite_superior, df['monto_usd_clean'])

    print("✅ PIPELINE COMPLETADO EXITOSAMENTE.")
    print("="*80)
    return df

# Ejecutar el pipeline de saneamiento

# 🏆 PASO 3: CAPA GOLD (DATA MART ANALYTICS & MACHINE LEARNING)
# ==============================================================================
def ejecutar_capa_gold(df_silver_in: pd.DataFrame) -> dict:
    print("="*80)
    print("🌟 INICIANDO GENERACIÓN DE CAPA GOLD (BUSINESS ANALYTICS & ML)")
    print("="*80)
    
    df_gold = df_silver_in.copy()
    
    # --------------------------------------------------------------------------
    # 1. AGREGACIONES EJECUTIVAS Y METRICAS CORE (DATA MART DE GASTOS)
    # --------------------------------------------------------------------------
    # Extracción de características temporales
    df_gold['anio_mes'] = df_gold['fecha_gasto_clean'].dt.to_period('M')
    
    # Tabla Fact: Resumen Mensual por Concepto y Método de Pago
    data_mart_mensual = df_gold.groupby(['anio_mes', 'concepto_clean']).agg(
        total_monto_usd=('monto_usd_final', 'sum'),
        promedio_monto_usd=('monto_usd_final', 'mean'),
        transacciones_count=('id_transaccion', 'count')
    ).reset_index()
    
    print("\n📊 [DATA MART GOLD] Resumen Ejecutivo por Concepto (Top 5 Filas):")
    display(data_mart_mensual.head(5))
    
    # --------------------------------------------------------------------------
    # 2. SEGMENTACIÓN DE COMPORTAMIENTO MEDIANTE K-MEANS CLUSTERING
    # --------------------------------------------------------------------------
    # Normalización de variables continuas para K-Means
    features_cluster = ['monto_usd_final', 'z_score_robusto']
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(df_gold[features_cluster])
    
    # Aplicación de K-Means (k=3 clusters: Gasto Bajo, Medio, Alto)
    kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
    df_gold['cluster_gasto'] = kmeans.fit_predict(X_scaled)
    
    # Mapeo de clusters a nombres interpretables por negocio
    cluster_means = df_gold.groupby('cluster_gasto')['monto_usd_final'].mean().sort_values()
    cluster_map = {old_label: f"Perfil_{idx+1}" for idx, old_label in enumerate(cluster_means.index)}
    df_gold['categoria_gasto_ml'] = df_gold['cluster_gasto'].map(cluster_map)
    
    # --------------------------------------------------------------------------
    # 3. DETECCIÓN AVANZADA DE ANOMALÍAS CON ISOLATION FOREST
    # --------------------------------------------------------------------------
    iso_forest = IsolationForest(contamination=0.05, random_state=42)
    # -1 representa anomalía, 1 representa transacción normal
    df_gold['anomalia_iforest'] = iso_forest.fit_predict(X_scaled)
    df_gold['es_anomalia_ml'] = np.where(df_gold['anomalia_iforest'] == -1, "ALERTA", "NORMAL")
    
    anomalias_detectadas = (df_gold['es_anomalia_ml'] == "ALERTA").sum()
    print(f"\n🤖 [MACHINE LEARNING] Anomalías operativas aisladas por Isolation Forest: {anomalias_detectadas} registros.")
    
    print("\n✅ CAPA GOLD GENERADA CON ÉXITO.")
    print("="*80)
    
    return {
        "df_gold_final": df_gold,
        "data_mart_mensual": data_mart_mensual
    }

# Ejecutar el Pipeline de la Capa Gold
resultados_gold = ejecutar_capa_gold(df_silver)
